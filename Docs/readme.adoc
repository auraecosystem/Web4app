⚙️ 3. UNIFIED RUNTIME CORE (

```js
  export class Web4Runtime {
  constructor() {
    this.state = {};
    this.modules = {};
    this.events = [];
  }
  // ======================
  // 🧠 KERNEL STATE
  // ======================
  set(path, value) {
    const keys = path.split(".");
    let obj = this.state;
    for (let i = 0; i < keys.length - 1; i++) {
      obj[keys[i]] = obj[keys[i]] || {};
      obj = obj[keys[i]];
    }
    obj[keys[keys.length - 1]] = value;
    this.emit("state:update", { path, value });
  }
  get(path) {
    return path.split(".").reduce((a, k) => a?.[k], this.state);
  }
  // ======================
  // ⚡ EVENT BUS
  // ======================
  emit(type, payload) {
    const event = { type, payload, time: Date.now() };
    this.events.push(event);
    window.dispatchEvent(new CustomEvent(type, { detail: payload }));
  }
  on(type, fn) {
    window.addEventListener(type, e => fn(e.detail));
  }
  // ======================
  // 🔌 MODULE SYSTEM
  // ======================
  register(name, module) {
    this.modules[name] = module;
    module.init?.(this);
  }
  run(name, input) {
    return this.modules[name]?.execute?.(input, this);
  }
}
```

export const XSIM = {
  execute(input, runtime) {
    let state = input.initial || 0;
    const steps = [];
    for (let i = 0; i < 5; i++) {
      state = state * 1.1 + 2;
      steps.push(state);
    }
    return {
      final: state,
      steps
    };
  }
};
